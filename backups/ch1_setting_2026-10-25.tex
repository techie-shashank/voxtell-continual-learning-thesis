% Card row (images on one baseline): I0 at the origin; I1, I2 right of it; dots node D; I6.
%   Each card Cj = fit around title Tj (above Ij), image Ij, prompt Pj (below Ij, two lines).
%   Connections (straight, horizontal at the image row, east -> west): C0->C1, C1->C2, C2->D, D->C6.
% Brace above C1..C6 (north west of C1 to north east of C6), label above it.
% Requirements row below the cards: heading H (left-aligned with C0), boxes R1 R2 R3 (5 mm gaps,
%   width computed so that R1.west = C0.west and R3.east = C6.east),
%   no connections (the requirements hold after every task).
\documentclass[12pt,tikz,border=2pt]{standalone}
\input{preamble}
\begin{document}
\begin{tikzpicture}[
    font=\footnotesize,
    img/.style={inner sep=0pt, outer sep=0pt},
    ttl/.style={inner sep=0.5mm, text height=1.6ex, text depth=0.4ex},
    prm/.style={inner sep=0.5mm, align=center, text width=\imgw, text height=1.6ex},
    card/.style={draw=black!70, line width=0.5pt, rounded corners=1pt, inner sep=1.2mm},
    flow/.style={-{Stealth[length=2mm]}, line width=0.5pt, black!70},
    req/.style={draw=black!70, line width=0.5pt, rounded corners=1pt, fill=voxgreen!20,
                align=flush left, inner sep=1.5mm},
  ]
\def\imgw{28mm}
\def\cgap{10.5mm}   % between images of neighbouring cards (card padding + arrow)

% Card row
\node[img] (I0) {\includegraphics[width=\imgw]{ch1_lung.png}};
\node[img, right=\cgap of I0] (I1) {\includegraphics[width=\imgw]{ch1_task_lateral_tibial.png}};
\node[img, right=\cgap of I1] (I2) {\includegraphics[width=\imgw]{ch1_task_medial_tibial.png}};
\node[right=7.5mm of I2, inner sep=0.5mm] (D) {\dots};
\node[img, right=7.5mm of D] (I6) {\includegraphics[width=\imgw]{ch1_task_lateral_meniscus.png}};
\foreach \j/\t/\p in {0/pretrained/{VoxTell segments,\\e.g.\ ``lung''},
                      1/task 1/{``lateral tibial\\cartilage''},
                      2/task 2/{``medial tibial\\cartilage''},
                      6/task 6/{``lateral\\meniscus''}} {
  \node[ttl, above=0.8mm of I\j] (T\j) {\t};
  \node[prm, below=0.8mm of I\j] (P\j) {\strut\p\strut};   % struts: equal height of all prompts
  \node[card, fit=(T\j)(I\j)(P\j)] (C\j) {};
}
% horizontal at the height of the images (all images share one baseline)
\draw[flow] (C0.east |- I0) -- (C1.west |- I1);
\draw[flow] (C1.east |- I1) -- (C2.west |- I2);
\draw[flow] (C2.east |- D) -- (D.west);
\draw[flow] (D.east) -- (C6.west |- D);

% Brace over the tasks
\draw[decorate, decoration={brace, amplitude=1.5mm}, line width=0.5pt, black!70]
  ([yshift=1.5mm]C1.north west) -- ([yshift=1.5mm]C6.north east)
  node[midway, above=2mm, inner sep=0.5mm] {new structures, one after another, each with $k$ labelled volumes};

% Requirements
% three equal boxes spanning exactly from C0.west to C6.east, 5 mm gaps, three lines each
\node[anchor=north west, inner sep=0.5mm] (H) at ([yshift=-5mm]C0.south west)
  {after each task, the same model has to segment};
\path let \p1 = (C0.west), \p2 = (C6.east), \n1 = {(\x2 - \x1 - 10mm)/3 - 3mm} in
  node[req, text width=\n1, anchor=north west] (R1) at ([yshift=-1.5mm]H.south west)
    {the structure\\just learned\\[0.3ex]$\to$ adaptation}
  node[req, text width=\n1, anchor=north west] (R2) at ([xshift=5mm]R1.north east)
    {the structures\\learned before\\[0.3ex]$\to$ forgetting}
  node[req, text width=\n1, anchor=north west] (R3) at ([xshift=5mm]R2.north east)
    {what it could segment\\before, e.g.\ the lung\\[0.3ex]$\to$ loss of pretrained ability};
\end{tikzpicture}
